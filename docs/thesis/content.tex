\chapter{Introduction}
\label{sec:introduction}

% P1: scope and specific problem
\begin{itemize}
  \item Coding agents write whole features; the developer's role shifts from writing to overseeing.
  \item Oversight is sparse and late: small chunks, little monitoring, green tests as proof \citep{dhanorkar_human_2026}; approval rises, scrutiny falls \citep{yu_habituation_2026}.
  \item Problem: \emph{when} to look. A review at the end shows a large diff next to green tests; interrupting at every edit shows half-written code.
\end{itemize}

% P2: what others did, why it matters
\begin{itemize}
  \item Interruptions are cheaper at subtask boundaries \citep{iqbal_leveraging_2006,adamczyk_if_2004}.
  \item Reviewers need understanding and rationale; small cohesive changes \citep{bacchelli_expectations_2013,tao_how_2012,rigby_convergent_2013}.
  \item Agents lack principled control points \citep{wang_humans_2026}; developers miss even sabotage \citep{ye_coding_2026}.
  \item No controlled study of where in an agent's work a human check should happen.
\end{itemize}

% P3: what I did, what is new
\begin{itemize}
  \item Idea: pause at semantic milestones.
  \item A milestone criterion with a decision test (\autoref{chap:criteria}).
  \item Within-participants study, N = 30: milestone pause vs.\ pause elsewhere vs.\ review at the end (\autoref{chap:study}).
  \item Live agent in opencode, held at a chosen point by a checkpoint tool.
  \item Contributions: the criterion, controlled comparison, reproducible harness.
\end{itemize}

\section{Pair Programming}
\begin{itemize}
  \item Talk is about blocks of the program \citep{bryant_pair_2008}.
  \item Knowledge transfer \citep{plonka_knowledge_2015}; re-synchronising understanding \citep{zieris_observations_2016}.
  \item None gives triggers for when to intervene.
\end{itemize}

\section{Code Review and Understanding}
\begin{itemize}
  \item Understanding is central \citep{bacchelli_expectations_2013,sadowski_modern_2018}.
  \item Small independent changes \citep{rigby_convergent_2013}; rationale, decomposition \citep{tao_how_2012}.
  \item Decomposition and ordering experiments \citep{dibiase_effects_2019,baum_associating_2019}.
  \item Review as decision-making \citep{gullstrand_code_2026}; reliability of comprehension proxies \citep{arvan_reliability_2026}.
\end{itemize}

\section{Timing of Interruptions}
\begin{itemize}
  \item Cheaper at subtask boundaries \citep{iqbal_leveraging_2006}.
  \item Coarse breakpoints less annoying, no effect on task time \citep{adamczyk_if_2004}.
  \item Converse question here: which moment gives the most to understand.
\end{itemize}

\section{Cognitive Forcing and Verification}
\begin{itemize}
  \item Cognitive forcing reduces overreliance \citep{bucinca_trust_2021,ghosh_experimental_2026,li_guided_2026}.
  \item Explanations help only if they enable verification \citep{fok_search_2024,zhang_you_2024}.
  \item The behaviour question is such a verification prompt, the same in all conditions.
\end{itemize}

\section{Ambiguous Requirements and Tests as Proof}
\begin{itemize}
  \item LLMs generate from unclear requirements without asking \citep{mu_clarifygpt_2024,yang_clarity_2026}.
  \item Generated code is coupled and bloated \citep{kravchuk_modular_2025,zhu_ai_2026}.
  \item Coverage is a weak proxy \citep{inozemtseva_coverage_2014}; mutation testing \citep{jia_analysis_2011}.
\end{itemize}

\section{Measuring Reliance}
\begin{itemize}
  \item Reliance vs.\ decision quality \citep{schoffer_ai_2025,guo_decision_2024}.
  \item Signal detection, \(d'\) and \(c\) \citep{stanislaw_calculation_1999,decarlo_signal_1998}.
\end{itemize}

\section{Gap}
\begin{itemize}
  \item No controlled study of where in an agent's work a human check should happen, or of its effect on understanding.
\end{itemize}


\chapter{Semantic Milestones}
\label{chap:criteria}
% [docs/archive/criteria-derivation.md, docs/study/expose.md]

\section{Subtasks as the Unit of Review}
\begin{itemize}
  \item Agent loop has no natural step; subtask = one sub-goal ending in a reviewable state.
  \item Milestone = review gate at such a boundary.
\end{itemize}

\section{Derivation}
\begin{itemize}
  \item Taxonomy method \citep{nickerson_method_2013}; meta-characteristic.
  \item Requirements R1--R5; 14 sources read in full; rejected candidates.
  \item First three candidate criteria (unspecified case, foundation fixed, test blind spot).
  \item Fused into one: the first two never occurred apart in the material; the test blind spot cannot occur at a pause (tests come last).
\end{itemize}

\section{The Criterion}
\begin{itemize}
  \item Definition: a subtask that settles a case the user story leaves open, in a decision that code still to be written builds on.
  \item Decision test: open case settled; built upon (a later file calls it, tests do not count); not yet built upon.
  \item Judged from the reviewer's view: story, plan, diff, visible tests; the agent's instructions do not count.
  \item Webhook example.
  \item Why the open case \citep{mu_clarifygpt_2024,yang_clarity_2026,dhanorkar_human_2026,zieris_observations_2016}; why before others build on it \citep{kravchuk_modular_2025,zhu_ai_2026,tao_how_2012,iqbal_leveraging_2006}.
\end{itemize}

\section{Application to the Study Material}
\begin{itemize}
  \item One milestone per feature, two each at subtask 1, 2, 3; the open case each settles.
  \item Justifies the placement; not a factor.
\end{itemize}

\chapter{Study Design}
\label{chap:study}
% [docs/study/expose.md, docs/study/analysis-plan.md, docs/harness/technical.md]

\section{Design}
\begin{itemize}
  \item RQ; within-participants; regime (BATCH, MILESTONE, NON-MILESTONE) \(\times\) variant (correct, flawed).
  \item Williams design, six groups \citep{williams_experimental_1949}; N = 30, 180 trials, 720 items.
  \item Review sizes per regime; practice feature not analysed.
\end{itemize}

\section{Materials}
\begin{itemize}
  \item Six user stories in one Next.js shop, four subtasks each, fresh copy per trial.
  \item Flaw = wrong design decision in the agent's instructions; half in the milestone.
  \item Table: features, milestones, open case, flaw subtask, wrong decision.
\end{itemize}

\section{Apparatus}
\begin{itemize}
  \item opencode web app, study build \citep{opencode_2026}; live model \citep{deepseek_2026}.
  \item Drehbuch added to the system prompt by the study server, hidden from agent and participant.
  \item Checkpoint tool: called in every regime, held at the review point.
  \item Rejection: agent asks, participant answers once, agent fixes.
  \item Container, sandboxed user; next feature planned in the background.
\end{itemize}

\section{Procedure}
\begin{itemize}
  \item Consent, demographics, NCS-6 \citep{coelho_very_2018}, S-TIAS \citep{mcgrath_measuring_2025}, practice feature.
  \item Per feature: plan, approve, review, accept/reject, behaviour question, finish, understanding questions, Paas \citep{paas_training_1992}.
  \item Preference ranking, debrief; about 60 minutes.
\end{itemize}

\section{Measurements}
\begin{itemize}
  \item Understanding: 4 items per feature, graded on the agent's code.
  \item Decisions, reversals, behaviour question.
  \item Confidence per item, mental effort per feature.
  \item Review behaviour telemetry; model time and tokens.
\end{itemize}

\section{Analysis}
\begin{itemize}
  \item Mixed models \citep{bates_fitting_2015}; contrasts M vs.\ B, M vs.\ N; Holm \citep{holm_simple_1979}.
  \item H1 understanding of the milestone (primary); H2 \(d'\); H3 overconfidence \citep{kuznetsova_lmertest_2017}; H4 reversals.
  \item Manipulation check: did the flaw land.
  \item Power by simulation: 18 points \(\to\) 43\,\% at N = 30.
\end{itemize}

\section{Pilot}
\begin{itemize}
  \item Item difficulty, guessability, time; agent adherence, flaws landing, gradeability, waiting.
  \item Test runs so far: full adherence; flaws landed 15/15 outside US1, about two thirds in US1.
\end{itemize}

\section{Limitations}
\begin{itemize}
  \item Milestones placed by the author.
  \item Real but instructed agent: code varies, flaws may not land.
  \item BATCH confounds timing with amount of code.
  \item One fused criterion: open case and dependency cannot be separated.
  \item Small project, lab setting.
\end{itemize}
